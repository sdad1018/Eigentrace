   144	def top_unsaid(vec, V, words, said_stems, k, extra_drop=()):
   145	    """Top-k vocab words near vec that the group did not say (unstarred)."""
   146	    scores = V @ vec
   147	    order = np.argsort(-scores)
   148	    out, seen = [], set()
   149	    for i in order[:2000]:
   150	        w = words[i]
   151	        if len(w) < 4 or w in extra_drop:
   152	            continue
   153	        st = porter_stem(w.split()[0]) if " " in w else porter_stem(w)
   154	        if st in said_stems or st in seen:
   155	            continue
   156	        seen.add(st)
   157	        out.append(w)
   158	        if len(out) >= k:
   159	            break
   160	    return out

   165	def arm_b(void_word, headline, h_vec, sent_vecs, E, V, words, HARD,
   166	          said_all_stems, top_k=5):
   167	    v = E(f"{void_word} in the context of {headline}")[0]
   168	    d = v - h_vec
   169	    n = np.linalg.norm(d)
   170	    if n < 1e-8:
   171	        return []
   172	    d = d / n
   173	    w_depth = float((v - h_vec) @ d)          # == n
   174	    proj = (V - h_vec) @ d                     # depth of every concept
   175	    cone = V @ v                               # closeness to the void word
   176	    pool = np.argsort(-cone)[:ARMB_POOL]
   177	    rows = []
   178	    for i in pool:
   179	        w = words[i]
   180	        if len(w) < 4 or w in HARD:
   181	            continue
   182	        if proj[i] < DEPTH_FRAC * w_depth:     # not past (~)w -- skip
   183	            continue
   184	        st = porter_stem(w.split()[0]) if " " in w else porter_stem(w)
   185	        if st == porter_stem(void_word) or st in said_all_stems:
   186	            continue
   187	        conv = int(np.sum(sent_vecs @ V[i] > CONV_THRESH))
   188	        if conv < CONV_MIN:
   189	            continue
   190	        rows.append((w, conv, float(proj[i])))
   191	    rows.sort(key=lambda r: (-r[1], -r[2]))
   192	    return [r[0] for r in rows[:top_k]]

   197	def stem_jaccard(a, b):
   198	    A = {porter_stem(x.split()[0]) for x in a}
   199	    B = {porter_stem(x.split()[0]) for x in b}
   200	    return len(A & B) / max(1, len(A | B))

   203	def field_cos(a, b, E):
   204	    if not a or not b:
   205	        return float("nan")
   206	    va = E(list(a)).mean(0)
   207	    vb = E(list(b)).mean(0)
   208	    va /= np.linalg.norm(va) + 1e-8
   209	    vb /= np.linalg.norm(vb) + 1e-8
   210	    return float(va @ vb)
